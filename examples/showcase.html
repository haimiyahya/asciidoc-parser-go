Introduction

The asciidoc-parser-go project is a native Go implementation that targets near-full compliance with the https://asciidoc.org/[AsciiDoc Language Specification] while maintaining compatibility with https://asciidoctor.org/[Asciidoctor].

using System;
using System.Collections.Generic;
using System.Linq;

namespace AsciiDocParser
{
    public class Document
    {
        public string Title { get; set; }
        public List<Block> Blocks { get; set; } = new();

        public void AddBlock(Block block)
        {
            Blocks.Add(block);
        }

        public override string ToString()
        {
            return $"Document: {Title} ({Blocks.Count} blocks)";
        }
    }

    public abstract class Block
    {
        public abstract void Render();
    }

    public class ParagraphBlock : Block
    {
        public string Content { get; set; }

        public override void Render()
        {
            Console.WriteLine($"<p>{Content}</p>");
        }
    }
}


// Async function to fetch user data
async function fetchUserData(userId) {
    try {
        const response = await fetch(`/api/users/${userId}`);

        if (!response.ok) {
            throw new Error(`HTTP error! status: ${response.status}`);
        }

        const userData = await response.json();

        // Transform the data
        const user = {
            id: userData.id,
            fullName: `${userData.firstName} ${userData.lastName}`,
            email: userData.email.toLowerCase(),
            isAdmin: userData.roles?.includes('admin') || false
        };

        return user;
    } catch (error) {
        console.error('Failed to fetch user:', error);
        throw error;
    }
}

// Usage with arrow functions and promises
const getUsers = async () => {
    const users = await Promise.all([
        fetchUserData(1),
        fetchUserData(2),
        fetchUserData(3)
    ]);
    return users.filter(u => u.isAdmin);
};

from dataclasses import dataclass
from typing import List, Optional
from datetime import datetime

@dataclass
class User:
    """Represents a user in the system."""
    id: int
    username: str
    email: str
    created_at: datetime
    is_active: bool = True

class UserService:
    """Service for managing users."""

    def __init__(self, db_connection):
        self.db = db_connection
        self._cache = {}

    async def get_user(self, user_id: int) -> Optional[User]:
        """Retrieve a user by ID with caching."""
        # Check cache first
        if user_id in self._cache:
            return self._cache[user_id]

        # Query database
        query = "SELECT * FROM users WHERE id = %s"
        result = await self.db.fetch_one(query, user_id)

        if result:
            user = User(**result)
            self._cache[user_id] = user
            return user

        return None

    async def create_user(self, username: str, email: str) -> User:
        """Create a new user."""
        query = """
            INSERT INTO users (username, email, created_at)
            VALUES (%s, %s, %s)
            RETURNING *
        """
        result = await self.db.fetch_one(
            query,
            username,
            email,
            datetime.now()
        )
        return User(**result)


use std::collections::HashMap;

#[derive(Debug, Clone)]
struct Document {
    title: String,
    blocks: Vec<Block>,
    metadata: HashMap<String, String>,
}

#[derive(Debug, Clone)]
enum Block {
    Paragraph { content: String },
    List { items: Vec<String> },
    Table { rows: Vec<Vec<String>> },
}

impl Document {
    fn new(title: impl Into<String>) -> Self {
        Document {
            title: title.into(),
            blocks: Vec::new(),
            metadata: HashMap::new(),
        }
    }

    fn add_block(&mut self, block: Block) {
        self.blocks.push(block);
    }

    fn block_count(&self) -> usize {
        self.blocks.len()
    }

    fn render(&self) -> String {
        let mut output = format!("# {}\n\n", self.title);

        for block in &self.blocks {
            match block {
                Block::Paragraph { content } => {
                    output.push_str(&format!("{}\n\n", content));
                }
                Block::List { items } => {
                    for item in items {
                        output.push_str(&format!("* {}\n", item));
                    }
                    output.push('\n');
                }
                Block::Table { rows } => {
                    // Simplified table rendering
                    for row in rows {
                        output.push_str(&format!("| {}\n", row.join(" | ")));
                    }
                    output.push('\n');
                }
            }
        }

        output
    }
}

fn main() {
    let mut doc = Document::new("My Document");

    doc.add_block(Block::Paragraph {
        content: "This is a sample paragraph.".to_string(),
    });

    doc.add_block(Block::List {
        items: vec![
            "First item".to_string(),
            "Second item".to_string(),
            "Third item".to_string(),
        ],
    });

    println!("{}", doc.render());
}
